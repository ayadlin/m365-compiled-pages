Installation

Debian/Ubuntu (APT Repository):

# Add GPG key
curl -fsSL https://ayadlin.github.io/m365-compiled-pages/KEY.gpg | sudo gpg --dearmor -o /usr/share/keyrings/m365.gpg

# Add repository
echo "deb [signed-by=/usr/share/keyrings/m365.gpg] https://ayadlin.github.io/m365-compiled-pages stable main" | sudo tee /etc/apt/sources.list.d/m365-compiled.list

# Install
sudo apt update && sudo apt install m365-compiled
